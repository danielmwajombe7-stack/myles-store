<?php
/*
 * Load all records only for the dashboard statistics.
 * The actual table still uses the paginated data provider.
 */
$allModels = $dataProvider->query->all();

$total = count($allModels);
$active = 0;
$inactive = 0;
$fault = 0;

foreach ($allModels as $meter) {

    $meterStatus = $meter->getStatusValue();

    $status = strtolower(
        trim((string) $meterStatus)
    );

    if ($status === 'active') {
        $active++;
    }

    if ($status === 'inactive') {
        $inactive++;
    }

    if (
        in_array(
            $status,
            ['fault', 'faulty', 'error'],
            true
        )
    ) {
        $fault++;
    }
}

                            && $model->$field !== null
